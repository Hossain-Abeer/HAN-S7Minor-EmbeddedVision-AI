    """
    5.3 VERIFY STRATIFIED SPLITTING

    Reads splits.csv. Prints a per-role / per-class count matrix.
    Flags:
      - Zero cells (hard failure for val/test)
      - Below hard floor: train < 5, val/test < 1
      - Below soft target: val/test < 3

    Design reference: designtracker-1.4.2

    Run from repo root:
        python src/verify_split.py

    Output:
      Console matrix
      logs/1.4.2_stratification.txt (via Tee)
    """

    import os
    import csv
    import sys

    import config
    import constants as C

    HARD_FLOOR_TRAIN   = 5
    HARD_FLOOR_VALTEST = 1
    SOFT_TARGET_VALTEST = 3
    ROLES = ("train", "val", "test")


    def _tee_stdout(log_path):
        os.makedirs(os.path.dirname(log_path), exist_ok=True)
        log_file = open(log_path, "w")

        class _Tee:
            def write(self, s):
                log_file.write(s)
                sys.__stdout__.write(s)

            def flush(self):
                log_file.flush()
                sys.__stdout__.flush()

        sys.stdout = _Tee()


    def load_splits(splits_csv):
        rows = []
        with open(splits_csv, "r", newline="") as f:
            reader = csv.DictReader(f)
            for r in reader:
                rows.append((r["filename"], r["role"]))
        return rows


    def class_from_filename(rel_path):
        prefix = rel_path.split("/")[0]
        try:
            return int(prefix)
        except ValueError:
            return None


    def build_matrix(rows, class_ids):
        matrix = {role: {cid: 0 for cid in class_ids} for role in ROLES}
        for rel_path, role in rows:
            cid = class_from_filename(rel_path)
            if cid is None or cid not in class_ids:
                continue
            if role not in matrix:
                continue
            matrix[role][cid] += 1
        return matrix


    def print_matrix(matrix, class_ids):
        header = f"{'class':>6} | " + " | ".join(f"{r:>6}" for r in ROLES)
        print(header)
        print("-" * len(header))
        for cid in class_ids:
            line = f"{cid:>6} | " + " | ".join(
                f"{matrix[r][cid]:>6}" for r in ROLES
            )
            print(line)
        print("-" * len(header))
        totals = {r: sum(matrix[r].values()) for r in ROLES}
        print(f"{'total':>6} | " + " | ".join(f"{totals[r]:>6}" for r in ROLES))


    def check_floors(matrix, class_ids):
        hard_failures = []
        soft_failures = []
        for role in ROLES:
            for cid in class_ids:
                n = matrix[role][cid]
                if role == "train":
                    if n < HARD_FLOOR_TRAIN:
                        hard_failures.append(
                            (role, cid, n, f"train < {HARD_FLOOR_TRAIN}")
                        )
                else:
                    if n < HARD_FLOOR_VALTEST:
                        hard_failures.append(
                            (role, cid, n,
                             f"{role} < {HARD_FLOOR_VALTEST} (zero cell)")
                        )
                    elif n < SOFT_TARGET_VALTEST:
                        soft_failures.append(
                            (role, cid, n,
                             f"{role} < {SOFT_TARGET_VALTEST} (soft target)")
                        )
        return hard_failures, soft_failures


    def report_failures(hard_failures, soft_failures):
        print()
        if not hard_failures and not soft_failures:
            print("All roles meet hard floors and soft targets. "
                  "Proceed to 6.1.")
            return

        if hard_failures:
            print(f"HARD FAILURES ({len(hard_failures)}):")
            for role, cid, n, reason in hard_failures:
                print(f"  {role:>6} class {cid:>2}: n={n:>2} ({reason})")
            print()
            print("Note: FRAME_STRIDE is already 1 under the motion-gated")
            print("architecture. Standard escalation (stride 5->3) is a")
            print("non-lever. Options:")
            print("  1. Re-run assign_roles with rare-class preference.")
            print("  2. Narrow classifier scope to fully-covered classes.")
            print("  3. Document as limitation and proceed.")

        if soft_failures:
            print()
            print(f"SOFT TARGET MISSES ({len(soft_failures)}):")
            for role, cid, n, reason in soft_failures:
                print(f"  {role:>6} class {cid:>2}: n={n:>2} ({reason})")
            print()
            print("Soft misses are NON-BLOCKING. Document in Preliminary ML §3")
            print("as a known limitation (thin val/test cells for those classes).")


    def main():
        log_path = os.path.join(config.LOGS_DIR, "1.4.2_stratification.txt")
        _tee_stdout(log_path)

        rows = load_splits(config.SPLITS_CSV)
        print(f"[INFO] read {len(rows)} rows from {config.SPLITS_CSV}")

        matrix = build_matrix(rows, C.CLASS_IDS)
        print("\nPer-role / per-class count matrix:")
        print_matrix(matrix, C.CLASS_IDS)

        hard_failures, soft_failures = check_floors(matrix, C.CLASS_IDS)
        report_failures(hard_failures, soft_failures)


    if __name__ == "__main__":
        main()

    def _tee_stdout(log_path):
        [ensure LOGS_DIR exists]
        [open log_file in write mode]
        class _Tee:
            def write(self, s):
                [log_file.write(s)]
                [sys.__stdout__.write(s)]   # prevents recursion
            def flush(self):
                [flush both]
        [swap sys.stdout = _Tee()]

    PARAMS:
      [LOG_PATH]   config.LOGS_DIR / "1.4.2_stratification.txt"

    DESCRIPTION:
      Same Tee pattern as 4.3 audit.py and 1.7.1 clean.py. Uses

    def load_splits(splits_csv):

        [open splits_csv for reading]
        [reader = csv.DictReader]
        for [r] in reader:

8.3 class_from_filename([REL_PATH]) -> int | None

    def class_from_filename(rel_path):
        [prefix = rel_path.split("/")[0]]
        try:
            return int(prefix)
        except ValueError:
            return None

    PARAMS:
      [REL_PATH]   one row's "filename" value,
                   e.g. "1/clip1_frame000003_ball1.png"

    DESCRIPTION:
      Class label lives in the parent-folder prefix. Same convention
      as 1.5.3 assemble.py: "1/" -> label 1.

      Why try/except: malformed prefixes (stray "unsorted/" path)
      return None. build_matrix then skips them silently - a bad row
      contributes to no cell, not to a wrong cell. Defensive without
      raising.


8.4 build_matrix([rows], [CLASS_IDS]) -> dict

    def build_matrix(rows, class_ids):
        [matrix = {role: {cid: 0 for cid in class_ids} for role in ROLES}]
        for [rel_path, role] in rows:
            [cid = class_from_filename(rel_path)]
            if [cid is None or cid not in class_ids]: continue
            if [role not in matrix]: continue
            [matrix[role][cid] += 1]
        return matrix

    PARAMS:
      [rows]         output of load_splits
      [CLASS_IDS]    constants.CLASS_IDS = [1..15]

    DESCRIPTION:
      Initializes ALL cells to 0 BEFORE counting. Critical: a class
      that doesn't appear in test at all still has entry
      test[class]=0, which makes zero-cell detection possible. If
      initialization were omitted, absent classes would KeyError, not
      count as 0.

      Three defence points per row: malformed prefix (None), class
      out of range, role not in ROLES. Each causes a silent skip
      rather than a corrupted count.

      Return shape:
        {
          "train": {1: 55, 2: 15, ..., 15: 29},
          "val":   {1: 22, 2:  5, ..., 15: 10},
          "test":  {1: 17, 2:  5, ..., 15:  4},
        }

    def print_matrix(matrix, class_ids):

        for [cid] in [class_ids]:
            [print f"{cid:>6} | {train:>6} | {val:>6} | {test:>6}"]

        [print totals row: total | T | V | Te]

      count matrix" - verification, not report figure. The

8.6 check_floors([matrix], [CLASS_IDS]) -> (list, list)

    def check_floors(matrix, class_ids):
        [hard_failures = []]
        [soft_failures = []]
        for [role] in ROLES:
            for [cid] in [class_ids]:
                [n = matrix[role][cid]]
                if role == "train":
                    if [n < HARD_FLOOR_TRAIN]:
                        [hard_failures.append((role, cid, n, reason))]
                else:
                    if [n < HARD_FLOOR_VALTEST]:
                        [hard_failures.append((role, cid, n, "zero cell"))]
                    elif [n < SOFT_TARGET_VALTEST]:
                        [soft_failures.append((role, cid, n, "soft"))]
        return hard_failures, soft_failures

    PARAMS:
      [matrix]       output of build_matrix
      [CLASS_IDS]    constants.CLASS_IDS

    THRESHOLDS (from devtracker 5.3 Verify line):
      [HARD_FLOOR_TRAIN]   = 5   - every class must have >=5 in train
                                    (thin CV folds otherwise)
      [HARD_FLOOR_VALTEST] = 1   - every class must have >=1 in val and
                                    test (zero cell = hard failure)
      [SOFT_TARGET_VALTEST] = 3  - val/test cells below 3 flagged but
                                    not blocking

    DESCRIPTION:
      Three-tier classification. Zero cells in val/test are HARD.
      Any cell below its floor is HARD. Cells in val/test between 1
      and 2 are SOFT - statistically noisy but not fatal.

      Why separate lists: the two failure types have different
      downstream consequences. Hard failures may require the
      scope-narrowing decision from 5.2's carry-forward. Soft misses
      need only a sentence in Preliminary §3. Bundling them into one
      list would obscure that difference.


8.7 report_failures([hard], [soft])

    def report_failures(hard_failures, soft_failures):
        if not hard and not soft:
            [print "All roles meet hard floors and soft targets.
                    Proceed to 6.1."]
            return
        if hard:
            [print "HARD FAILURES (n):"]
            [per-failure line: role class cid: n=<n> (reason)]
            [print escalation note: FRAME_STRIDE=1, three options]
        if soft:
            [print "SOFT TARGET MISSES (n):"]
            [per-failure line]
            [print non-blocking note]

    PARAMS:  outputs of check_floors.

    DESCRIPTION:
      Two-section report. Hard failures first - they demand an action.
      Soft misses second - they demand a sentence. The escalation note
      is hardcoded (not merely referenced) because a future reader of
      the log file won't have the devtracker at hand. It reads:

        "FRAME_STRIDE is already 1 under the motion-gated architecture.
         Standard escalation (stride 5->3) is a non-lever. Options:
           1. Re-run assign_roles with rare-class preference.
           2. Narrow classifier scope to fully-covered classes.
           3. Document as limitation and proceed."

      These three options are the fallback chain from 5.2 writeup §7.
      Recorded in the log file for traceability.

      check -> report. Print happens BEFORE check so that even if
      check crashes (it won't - but defensively), the matrix is
      already in the log for inspection.

9. DESIGN DECISIONS - ALL CONFIRMED
================================================================================

(a) Matrix initialization: {cid: 0} BEFORE counting.
    Rejected: lazy counting (misses zero cells, turns them into
    KeyError). Rationale: absent class in a role MUST appear as n=0 to
    trigger the hard-failure check.

(b) Class label from filename prefix.
    Rejected: read class from parent folder name on disk. Rationale:
    the CSV filename is the authoritative source; walking the disk
    again would allow divergence if files were added/moved after 5.1.

(c) Malformed rows silently skipped.
    Rejected: raise on malformed. Rationale: one bad row should not
    corrupt the audit; it will be caught as a missing file elsewhere.

(d) Log path: logs/1.4.2_stratification.txt.
    Matches devtracker 5.3 Save line. User renamed their copy to
    verify_split_log.txt - traceability note, not a bug.

(e) Tee helper consistent with 4.3 and 1.7.1.
    Rejected: plain print (log lost). Rejected: separate log file
    writing after the fact (easy to forget).

(f) Thresholds as module constants (HARD_FLOOR_TRAIN etc.).
    Rejected: hardcoded literals. Rationale: tunable in one place if
    a later step revises the floor (e.g. from 5 to 3 for train if CV
    folds prove noisy).

(g) Hard vs soft as separate lists and separate report sections.
    Rationale: different remediation paths (scope narrowing vs
    one-sentence limitation).

(h) Escalation note hardcoded in the log output.
    Rationale: log must be self-contained. A reader of the log six
    months later cannot be expected to cross-reference the
    designtracker for the "why stride 3 is dead" argument.
