import os

# ============================================================================

                                    #        billiards-game-clips

# ============================================================================
# CREATE CLASS FOLDERS
# ============================================================================
def ensure_class_folders(crop_root, class_ids):
    """
    Create 1/ ... 15/ subfolders under crop_root if missing.
    Idempotent: safe to call every run.
    """
    os.makedirs(crop_root, exist_ok=True)
    for cid in class_ids:
        os.makedirs(os.path.join(crop_root, str(cid)), exist_ok=True)


if __name__ == "__main__":
    ensure_class_folders(CROP_ROOT, CLASS_IDS)

4. CODE DESCRIPTION
--------------------------------------------------------------------------------
os.makedirs(crop_root, exist_ok=True)
  Creates parent if absent. exist_ok=True suppresses the FileExistsError
  that would otherwise fire on re-runs. Idempotent.

os.makedirs(os.path.join(crop_root, str(cid)), exist_ok=True)
  One folder per class. str(cid) is required because folder names are
  strings; int would raise TypeError.

Loop over class_ids (1..15)
  Cue is excluded by design: it has no numeral, so it is not an OCR class.
  Snooker is excluded by 0.4.2.

Why idempotent
  Extraction and crop runs will be repeated (rescale triggers at 1.2.3 and
  1.3.5). Folder creation must not fail on the second pass. Manually
  creating the folders once (as done here) is equivalent to running this
  function once.
